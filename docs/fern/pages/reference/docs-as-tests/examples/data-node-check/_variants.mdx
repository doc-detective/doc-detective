# Check your Node.js version

1. Print the Node.js version:

   ```shell
   node --version
   ```

   The output starts with `v` and the major version, such as `v22.11.0`. You need version 18 or later.
2. If Node.js printed a version, print the npm version:

   ```shell
   npm --version
   ```
3. Find the Node.js executable. On Windows, run:

   ```shell
   where node
   ```

   On macOS and Linux, run:

   ```shell
   command -v node
   ```

if:

tests:
  - testId: data-node-check
    if:
      - $$platform oneOf [linux, mac, windows]
    publicVariables:
      - NODE_MAJOR
    steps:
      - stepId: node-version
        runShell: node --version
        assertions: $$outputs.stdio.stdout matches /^v[0-9]+\./
        variables:
          NODE_MAJOR: extract($$stdio.stdout, "v([0-9]+)")
      - if:
          - $$steps.node-version.outputs.exitCode == 0
          - $$steps.node-version.outputs.stdio.stdout matches /^v/
        runShell: npm --version
        assertions:
          - $$outputs.exitCode == 0
          - $NODE_MAJOR >= 18
      - if: $$platform == windows
        runShell: where node
        onSkip:
          - continue: true
      - if: $$platform != windows
        runShell: command -v node
        onFail:
          - stop: test
